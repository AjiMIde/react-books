# ts config

* 学习`ts`的配置，最快速的方式是搜索，查看一些热门的配置，或是B站学习
* 最后是查看官方文档

## compilerOptions

```ts
const compilerOptions = {
  /* base options */
  /* base options */
  /* base options */
  // 1,允许将ts文件编译成d.ts文件?
  // 2,ts编译器会将项目视为支持基于项目的增量编译，在大型项目中有用，只编译受影响的文件（而非整个项目）
  // 简单项目没有必要开启
  // 3,composite 选项会强制执行某些约束，使得构建工具，可以快速确定一个工程是否已经建立。(from官方)
  "composite": true,
  // 存储增量编译信息的文件 与incremental、composite配合使用，增量编译，可以更快的编译
  "tsBuildInfoFile": "./node_modules/.tmp/tsconfig.app.tsbuildinfo",
  // 生成代码的目标语法版本
  "target": "ES2020",
  // 需要引入到编译中的库，比如 DOM.Iterable，如果不引用，你就不能对一个 dom.elements 进行 for 操作 
  "lib": [
    "ES2020",
    "DOM",
    "DOM.Iterable"
  ],
  // https://www.typescriptlang.org/docs/handbook/modules/theory.html#module-resolution
  // 引用库时的类型是如何的，可为 commonjs/ESNext/amd/umd/....
  "module": "ESNext",
  // 如何解析引用的模块 另外还有 node/nodenext等
  // https://www.typescriptlang.org/docs/handbook/modules/theory.html#module-resolution
  "moduleResolution": "bundler",
  // 跳过对某些库的d.ts的检查，提高编译效率，但只有编译器检查时报错
  "skipLibCheck": true,
  // 当它设置为true时,类的实例属性会编译成使用 Object.defineProperty，当 target 为 ES2020 and more，此为 true
  "useDefineForClassFields": true,
  // 用于配置模块解析策略，bundler时对应如 webpack/vite/rollup等模块解析的策略，会根据exports/imports解析
  // 允许TypeScript文件以.ts、.mts或.tsx等TypeScript特有的扩展名相互导入
  "allowImportingTsExtensions": true,
  // 识别JSON
  "resolveJsonModule": true,
  // isolatedModules 选项后，TypeScript 将会在当你写的某些代码不能被单文件转译的过程正确处理时警告你。此项默认true
  "isolatedModules": true,
  // 模块识别，一般为auto，会自行判断，也可为legacy/force，这个配置似乎意义不大
  "moduleDetection": "force",
  // 不生成文件，一般只用在IDE进行实行实时检查；
  "noEmit": true,
  // 支持 jsx格式， 'preserve', 'react-native', or 'react'
  "jsx": "react-jsx",
  /* Linting */
  /* Linting */
  /* Linting */
  // 严格模式
  "strict": true,
  // 1,(官方)当某个变量可能为 null/undefinded 时，调用其属性时，会报错，当strict:true时，此条为true
  // 2,不允许把 null、undefined 赋值给其他类型变量
  "strictNullChecks": true,
  // 不允许隐式地"表示" any 类型，当strict:true时，此条为true
  "noImplicitAny": true,
  // 使用call/bind/apply时，强制使用严格模式，当strict:true时，此条为true
  "strictBindCallApply": true,
  // 强制文件引入时区分大小写，此条默认为true
  "forceConsistentCasingInFileNames": true,
  // 未使用的变量是否报错，此条默认为true
  "noUnusedLocals": false,
  // 同上，未使用的参数是否报错，此条默认为true
  "noUnusedParameters": true,
  // break/return/throw必须出现在 switch/case 中，此条默认为true
  "noFallthroughCasesInSwitch": true,

  /* 其他 */
  /* 其他 */
  /* 其他 */
  //
  "baseUrl": ".",
  // 一般与打包工具一起配合设置别名
  "paths": {
    "@/*": ["src/*"],
    "@az": ["../../../fzs-amazing-builder/src"],
  }
}
```
